**A. Create `apps/web/features/landing/components/Hero.tsx`** — RSC, owns auth-aware CTA logic, English `font-serif` h1 / Vietnamese `font-sans` h1:

```tsx
import { getLocale, getTranslations } from 'next-intl/server';
import { cookies, headers } from 'next/headers';
import Link from 'next/link';

import { buttonVariants } from '@/components/ui/button';
import { getCurrentUserCached } from '@/features/account/api/me';
import { cn } from '@/lib/utils';

import type { AppLocale } from '@/i18n/routing';

export default async function Hero() {
  const t = await getTranslations();
  const locale = (await getLocale()) as AppLocale;

  let ctaHref = '/login';
  let ctaKey: 'landing.primaryCta' | 'landing.continueSetupCta' = 'landing.primaryCta';
  try {
    const headerStore = await headers();
    const cookieHeader = headerStore.get('cookie') ?? '';
    if (cookieHeader) {
      const user = await getCurrentUserCached(cookieHeader);
      if (user.onboardingStep && user.onboardingStep !== 'COMPLETE') {
        ctaHref = '/onboarding';
        ctaKey = 'landing.continueSetupCta';
      }
    }
  } catch {
    // Silent — default to /login. Privacy: never log cookie or response.
  }

  // Instrument Serif has no Vietnamese subset (RESEARCH §2 + Pitfall 2).
  // Use font-serif only for English; font-sans for Vietnamese.
  const headingFontClass = locale === 'en' ? 'font-serif' : 'font-sans';

  return (
    <section className="relative px-6 py-20 sm:py-28 lg:py-32">
      <div className="mx-auto flex w-full max-w-3xl flex-col items-start gap-8">
        <span className="bg-accent-soft text-accent inline-flex items-center gap-2 rounded-full px-3 py-1 text-xs font-medium">
          <span className="bg-accent size-1.5 rounded-full" />
          {t('landing.eyebrow')}
        </span>
        <h1 className={cn('text-foreground text-4xl leading-tight tracking-tight sm:text-5xl lg:text-6xl', headingFontClass)}>
          {t('landing.heading')}
        </h1>
        <p className="text-muted-foreground max-w-2xl text-lg leading-relaxed">
          {t('landing.tagline')}
        </p>
        <div className="flex flex-col items-start gap-3 sm:flex-row sm:items-center">
          <Link href={ctaHref} className={cn(buttonVariants({ size: 'lg' }))}>
            {t(ctaKey)}
          </Link>
        </div>
      </div>
    </section>
  );
}
```
