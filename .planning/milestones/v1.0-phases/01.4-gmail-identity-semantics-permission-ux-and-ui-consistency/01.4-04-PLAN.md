import { cn } from "@/lib/utils";

interface SectionCardProps extends React.ComponentProps<"div"> {
  heading?: string;
  description?: string;
  footer?: React.ReactNode;
}

export function SectionCard({ heading, description, footer, className, children, ...props }: SectionCardProps) {
  const showHeader = Boolean(heading || description);
  return (

      {showHeader && (
        <CardHeader>

          {description && <CardDescription>{description}</CardDescription>}
        </CardHeader>
      )}
      <CardContent>{children}</CardContent>
      {footer && <CardFooter>{footer}</CardFooter>}
    </Card>
  );
}

```tsx
'use client';

import * as React from "react";
import { cva, type VariantProps } from "class-variance-authority";
import { useTranslations } from "next-intl";
import { Info, CheckCircle2, AlertTriangle, AlertCircle } from "lucide-react";

import { Alert, AlertTitle, AlertDescription } from "@/components/ui/alert";
import { Button } from "@/components/ui/button";
import { cn } from "@/lib/utils";

const statusAlertVariants = cva("", {
  variants: {
    variant: {
      info:    "border-border bg-muted text-foreground",
      success: "border-border bg-muted text-foreground",
      warn:    "border-border bg-muted text-foreground",
      error:   "border-destructive/40 bg-destructive/10 text-destructive",
    },
  },
  defaultVariants: { variant: "info" },
});

const iconByVariant = {
  info:    Info,
  success: CheckCircle2,
  warn:    AlertTriangle,
  error:   AlertCircle,
} as const;

type StatusAlertVariant = keyof typeof iconByVariant;

interface StatusAlertProps extends VariantProps<typeof statusAlertVariants> {
  titleKey: string;
  bodyKey?: string;
  params?: Record<string, string | number | Date>;
  cta?: { labelKey: string; onClick?: () => void; href?: string };
  className?: string;
}

export function StatusAlert({ variant, titleKey, bodyKey, params, cta, className }: StatusAlertProps) {
  const t = useTranslations();
  const Icon = iconByVariant[(variant ?? "info") as StatusAlertVariant];
  return (
    <Alert role="alert" className={cn(statusAlertVariants({ variant }), className)} data-slot="status-alert">
      <Icon aria-hidden className="size-5" />
      <AlertTitle>{t(titleKey as never, params)}</AlertTitle>
      {bodyKey && <AlertDescription>{t(bodyKey as never, params)}</AlertDescription>}
      {cta && (
        <Button
          variant="outline"
          size="sm"
          onClick={cta.onClick}
          className="mt-2 w-fit"
          type="button"
        >
          {t(cta.labelKey as never)}
        </Button>
      )}
    </Alert>
  );
}
```

**LOCKED design choices** (from UI-SPEC + CONTEXT + Wave 0 test):
- Variant prop is a CLOSED enum: `'info' | 'success' | 'warn' | 'error'` only (per CONTEXT D-D3 "closed enum" + RESEARCH §Security Domain Threat T-error-param-tamper — UI never accepts arbitrary status strings).
- Props accept i18n KEYS only (`titleKey`, `bodyKey`, `cta.labelKey`) — NOT raw strings (per CONTEXT D-D3 + UI-SPEC §"Discretion items" recommendation: "strict `titleKey` / `bodyKey` only"). The `as never` cast lets TypeScript accept dynamic keys; runtime falls back to key path if missing (next-intl 4.x default, verified Phase 1.3 P05).
- Color mapping per UI-SPEC §Color "Status alert color mapping": info/success/warn use neutral `bg-muted` (Discretion: executor MAY add `--color-success`/`--color-warning` semantic tokens to globals.css if review finds neutrals too indistinguishable — DEFER unless review demands; per RESEARCH §Finding #3 the existing tokens are sufficient).
- `error` variant uses `border-destructive/40 bg-destructive/10 text-destructive` — the only variant with destructive accents (UI-SPEC §Color "Destructive ... reserved for ... error variant accents only").
- Icons from `lucide-react` (already in deps; verified). Each icon has `aria-hidden` because the variant is announced via the Alert's role="alert".
- CTA uses `<Button variant="outline" size="sm">` — keeps the alert visually quieter than the primary CTA elsewhere on the page.
- NEVER render `error.message` or any user-data string here. Privacy contract D-D3.
- The icon is a wrapper element, not a slot — keeping the API simple. Customization (different icon per use) deferred to Phase 5 if needed.

**Privacy:** No log statements, no console.* calls. The component is a pure render. If a consumer passes an i18n key whose value contains untrusted ICU param interpolation (e.g., user email), the consumer is responsible for sanitization — but per UI-SPEC §Copywriting Contract no current i18n key in this phase carries user PII.
