```typescript
"use client";
import { useTranslations, useMessages } from 'next-intl';
import type { components } from './schema';

export type ApiError = components['schemas']['ApiError'];
export type FieldError = components['schemas']['FieldErrorDto'];

function hasNestedKey(messages: unknown, dottedKey: string): boolean {
  let cur: any = messages;
  for (const part of dottedKey.split('.')) {
    if (cur == null || typeof cur !== 'object' || !(part in cur)) return false;
    cur = cur[part];
  }
  return typeof cur === 'string';
}

export function useLocalizedApiError() {
  const t = useTranslations('errors');
  const messages = useMessages() as Record<string, unknown>;
  return (err: ApiError | undefined): string => {
    if (!err?.code) return t('unknown');
    const code = err.code;
    // Render sentinel for missing keys instead of silent fallback (security_threat_model #6)
    if (!hasNestedKey(messages.errors, code)) {
      // production: localized 'unknown' AND visible sentinel for QA
      if (process.env.NODE_ENV !== 'production') return `[MISSING:${code}]`;
      return t('unknown');
    }
    try {
      return t(code as never, (err.params ?? {}) as never);
    } catch {
      return t('unknown');
    }
  };
}

export function useLocalizedFieldError() {
  const t = useTranslations('errors');
  const messages = useMessages() as Record<string, unknown>;
  return (fe: FieldError | undefined): string => {
    if (!fe?.code) return t('fieldFallback');
    if (!hasNestedKey(messages.errors, fe.code)) {
      if (process.env.NODE_ENV !== 'production') return `[MISSING:${fe.code}]`;
      return t('fieldFallback');
    }
    try {
      return t(fe.code as never, (fe.params ?? {}) as never);
    } catch {
      return t('fieldFallback');
    }
  };
}
```
