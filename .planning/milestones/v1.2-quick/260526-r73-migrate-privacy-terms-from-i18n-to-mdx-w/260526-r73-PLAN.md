Raw .md route handler shape (apps/web/app/(public)/privacy.md/route.ts):

```typescript
import { promises as fs } from 'node:fs';
import matter from 'gray-matter';
import { getLocale } from 'next-intl/server';
import { buildDocPath } from '@/lib/docs/loader';

const SLUG = 'privacy' as const; // or 'terms' in terms.md/route.ts

export async function GET() {
  const locale = await getLocale();
  if (locale !== 'vi' && locale !== 'en') {
    return new Response('Not Found', { status: 404 });
  }

  let raw: string;
  try {
    raw = await fs.readFile(buildDocPath(SLUG, locale), 'utf8');
  } catch {
    return new Response('Not Found', { status: 404 });
  }

  const { content } = matter(raw); // strips frontmatter
  return new Response(content, {
    status: 200,
    headers: {
      'content-type': 'text/markdown; charset=utf-8',
      'cache-control': 'public, max-age=300',
    },
  });
}
```
