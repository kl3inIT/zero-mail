  export function safeCount(value: number | undefined): number
  export function precisionRate(decisions: number | undefined, applied: number | undefined): string
    Returns "—" when safeCount(decisions) === 0
    Returns Math.round(safeCount(applied) / safeCount(decisions) * 100) + "%" otherwise
  export function trustBadgeVariant(decisions: number | undefined, applied: number | undefined):
    "default" | "secondary" | "destructive"
    Returns "secondary" when decisions === 0 (badge hidden/neutral)
    Returns "default" (teal) when pct >= 90
    Returns "secondary" (amber) when pct >= 70
    Returns "destructive" when pct < 70
