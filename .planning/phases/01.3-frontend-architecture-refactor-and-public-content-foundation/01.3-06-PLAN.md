// apps/web/app/(public)/docs/[slug]/page.tsx
import { promises as fs } from 'node:fs';
import path from 'node:path';
import { notFound } from 'next/navigation';
import { compileMDX } from 'next-mdx-remote/rsc';
import { getLocale, getTranslations } from 'next-intl/server';
import Link from 'next/link';

interface Frontmatter {
  title: string;
  slug: string;
  order: number;
  locale: 'vi' | 'en';
}

const SLUG_RE = /^[a-z0-9-]+$/;

export default async function DocPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  if (!SLUG_RE.test(slug)) notFound();

  const locale = await getLocale();
  if (locale !== 'vi' && locale !== 'en') notFound();

  const filePath = path.join(
    process.cwd(),
    'content',
    'docs',
    `${slug}.${locale}.mdx`,
  );

  let source: string;
  try {
    source = await fs.readFile(filePath, 'utf8');
  } catch {
    notFound();
  }

  const t = await getTranslations('docs');
  const { content, frontmatter } = await compileMDX<Frontmatter>({
    source,
    options: { parseFrontmatter: true },
    // NOTE: blockJS / blockDangerousJS defaults from v6 are intentionally NOT overridden.
  });

  return (
    <article className="prose mx-auto max-w-3xl px-4 py-8">
      <h1>{frontmatter.title}</h1>
      {content}
      <p className="mt-8">
        <Link href="/docs">{t('backToList')}</Link>
      </p>
    </article>
  );
}

<!-- Docs index — fs.readdir + gray-matter -->

```tsx
// apps/web/app/(public)/docs/page.tsx
import { promises as fs } from 'node:fs';
import path from 'node:path';
import matter from 'gray-matter';
import Link from 'next/link';
import { getLocale, getTranslations } from 'next-intl/server';

interface Frontmatter {
  title: string;
  slug: string;
  order: number;
  locale: 'vi' | 'en';
}

const FILENAME_RE = /^([a-z0-9-]+)\.(vi|en)\.mdx$/;

export default async function DocsIndexPage() {
  const t = await getTranslations('docs');
  const locale = await getLocale();
  const docsDir = path.join(process.cwd(), 'content', 'docs');

  let entries: string[];
  try {
    entries = await fs.readdir(docsDir);
  } catch {
    entries = [];
  }

  const docs: Frontmatter[] = [];
  for (const name of entries) {
    const m = name.match(FILENAME_RE);
    if (!m) continue;
    if (m[2] !== locale) continue;
    const raw = await fs.readFile(path.join(docsDir, name), 'utf8');
    const { data } = matter(raw);
    docs.push(data as Frontmatter);
  }
  docs.sort((a, b) => a.order - b.order);

  return (
    <section className="mx-auto max-w-3xl px-4 py-8">
      <h1 className="text-2xl font-semibold mb-6">{t('indexHeading')}</h1>
      {docs.length === 0 ? (
        <div>
          <p className="text-base font-semibold">{t('empty.heading')}</p>
          <p className="text-sm text-muted-foreground">{t('empty.body')}</p>
        </div>
      ) : (
        <ul className="space-y-2">
          {docs.map((d) => (
            <li key={d.slug}>
              <Link href={`/docs/${d.slug}`} className="text-primary underline">
                {d.title}
              </Link>
            </li>
          ))}
        </ul>
      )}
    </section>
  );
}
```

|-----------|----------|-----------|-------------|-----------------|

| T-1.3.06-02 | E (Elevation of privilege) | Untrusted MDX runs JS in RSC | mitigate | next-mdx-remote v6 ships `blockJS: true` and `blockDangerousJS: true` by DEFAULT. Acceptance criterion asserts these are NOT overridden. Phase 1.3 content is repo-owned, but this defense holds for future phases. |
| T-1.3.06-03 | I (Information disclosure) | fs error path leaks file paths | mitigate | All fs failures route to `notFound()` — no exception bubble; no path string in render output. |
</threat_model>
