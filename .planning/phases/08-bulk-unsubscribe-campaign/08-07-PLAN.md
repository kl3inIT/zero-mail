       @Service
       @Transactional(readOnly = true)
       public class CampaignPreviewService {
           private final JdbcTemplate jdbcTemplate;
           private final SenderSuppressionRepository senderSuppressionRepository;
           private final Clock clock;

           public CampaignPreviewService(JdbcTemplate jdbcTemplate, SenderSuppressionRepository senderSuppressionRepository, Clock clock) { ... }

           public CampaignPreviewResult preview(UUID tenantId, List<String> senderEmails) {
               // 1. Cap check: senderEmails.size() ≤ 25
               if (senderEmails.size() > UnsubscribeCampaignPolicy.MAX_SENDERS_PER_CAMPAIGN) {
                   throw new CampaignCapExceededException(Kind.TOO_MANY_SENDERS, senderEmails.size(), UnsubscribeCampaignPolicy.MAX_SENDERS_PER_CAMPAIGN);
               }

               // 2. Per-sender lookup: messageCount, unsubscribeMethod, suppressed flag
               List<PerSenderPreview> perSender = new ArrayList<>();
               int totalHistoryCount = 0;
               for (String senderEmail : senderEmails) {
                   PerSenderPreview row = lookupPerSenderPreview(tenantId, senderEmail);
                   perSender.add(row);
                   if (row.unsubscribeMethod() != UnsubscribeMethod.NONE && !row.suppressed()) {
                       totalHistoryCount += row.messageCount();
                   }
               }

               // 3. Cap check: total history ≤ 2000
               if (totalHistoryCount > UnsubscribeCampaignPolicy.MAX_HISTORY_MESSAGES_PER_CAMPAIGN) {
                   throw new CampaignCapExceededException(Kind.TOO_MANY_MESSAGES, totalHistoryCount, UnsubscribeCampaignPolicy.MAX_HISTORY_MESSAGES_PER_CAMPAIGN);
               }

               return new CampaignPreviewResult(perSender, totalHistoryCount);
           }

           private PerSenderPreview lookupPerSenderPreview(UUID tenantId, String senderEmail) {
               String senderDomain = senderEmail.substring(senderEmail.indexOf('@') + 1);

               // Check suppressed
               boolean suppressed = senderSuppressionRepository.findByTenantIdAndSenderEmail(tenantId, senderEmail).isPresent()
                   || senderSuppressionRepository.findByTenantIdAndSenderDomain(tenantId, senderDomain).isPresent();

               // Lookup unsubscribe method + count (30d window per SPEC UNS-01)
               Instant windowStart = clock.instant().minus(Duration.ofDays(30));
               Map<String, Object> row = jdbcTemplate.queryForMap(
                   "SELECT COUNT(*) AS msg_count, " +
                   "  BOOL_OR(list_unsubscribe_one_click) AS has_one_click, " +
                   "  BOOL_OR(list_unsubscribe_mailto IS NOT NULL) AS has_mailto " +
                   "FROM mail_message_observed " +
                   "WHERE tenant_id = ? AND sender_email = ? AND observed_at >= ?",
                   tenantId, senderEmail, Timestamp.from(windowStart)
               );
               long messageCount = ((Number) row.get("msg_count")).longValue();
               boolean hasOneClick = Boolean.TRUE.equals(row.get("has_one_click"));
               boolean hasMailto = Boolean.TRUE.equals(row.get("has_mailto"));
               UnsubscribeMethod method = hasOneClick ? UnsubscribeMethod.ONE_CLICK : (hasMailto ? UnsubscribeMethod.MAILTO : UnsubscribeMethod.NONE);
               return new PerSenderPreview(senderEmail, senderDomain, method, messageCount, suppressed);
           }

           public record CampaignPreviewResult(List<PerSenderPreview> perSender, int totalHistoryCount) {
               public CampaignPreviewResult { perSender = List.copyOf(perSender); }
           }

           public record PerSenderPreview(String senderEmail, String senderDomain, UnsubscribeMethod unsubscribeMethod, long messageCount, boolean suppressed) {
               public String riskBadge() {
                   if (suppressed) return "SUPPRESSED_BLOCKED";
                   if (unsubscribeMethod == UnsubscribeMethod.NONE) return "NO_HEADER_DISABLED";
                   return "SAFE";
               }
               public boolean willArchive() {
                   return unsubscribeMethod != UnsubscribeMethod.NONE && !suppressed;
               }
           }

       @Service
       public class CampaignExecuteService {
           private final UnsubscribeCampaignRepository unsubscribeCampaignRepository;
           private final UnsubscribeAttemptRepository unsubscribeAttemptRepository;
           private final ProcessingJobRepository processingJobRepository;
           private final CampaignPreviewService campaignPreviewService;
           private final ObjectMapper objectMapper;
           private final Clock clock;

           public CampaignExecuteService(... constructor inject all) { ... }

           @Transactional
           public CampaignExecuteResult execute(UUID tenantId, List<String> senderEmails) {
               // 1. Re-run preview validation (defense-in-depth — controller may have skipped)
               CampaignPreviewService.CampaignPreviewResult preview = campaignPreviewService.preview(tenantId, senderEmails);

               // 2. Filter out non-archivable (suppressed + NO_HEADER) — only includes willArchive() senders
               List<CampaignPreviewService.PerSenderPreview> archivable = preview.perSender().stream()
                   .filter(CampaignPreviewService.PerSenderPreview::willArchive)
                   .toList();

               if (archivable.isEmpty()) {
                   throw new IllegalStateException("No archivable senders in selection");
               }

               // 3. INSERT campaign
               UnsubscribeCampaignEntity campaign = new UnsubscribeCampaignEntity(
                   tenantId, archivable.size(), preview.totalHistoryCount());
               campaign = unsubscribeCampaignRepository.save(campaign);

               // 4. INSERT N attempts
               for (CampaignPreviewService.PerSenderPreview row : archivable) {
                   UnsubscribeAttemptEntity attempt = new UnsubscribeAttemptEntity(
                       campaign.getId(), row.senderEmail(), row.senderDomain(), row.unsubscribeMethod());
                   unsubscribeAttemptRepository.save(attempt);
               }

               // 5. INSERT processing_job với payload {"campaignId": uuid, "schemaVersion": 1} (D-19)
               String payload = objectMapper.writeValueAsString(new CampaignJobPayload(campaign.getId(), 1));
               ProcessingJobEntity job = new ProcessingJobEntity(tenantId, "UNSUBSCRIBE_CAMPAIGN", payload);
               job = processingJobRepository.save(job);

               // 6. Link job_id back to campaign (so worker can update campaign.jobId)
               campaign.linkJob(job.getId());
               unsubscribeCampaignRepository.save(campaign);

               log.info("event=cleanup_campaign_executed tenantId={} campaignId={} jobId={} senderCount={} historyCount={}",
                       tenantId, campaign.getId(), job.getId(), archivable.size(), preview.totalHistoryCount());

               return new CampaignExecuteResult(campaign.getId(), job.getId());
           }

           public record CampaignExecuteResult(UUID campaignId, UUID jobId) {}
           private record CampaignJobPayload(UUID campaignId, int schemaVersion) {}  // D-19
       }
       ```

    @Service
    public class CampaignUndoService {
        private static final Logger log = LoggerFactory.getLogger(CampaignUndoService.class);

        private final UnsubscribeCampaignRepository unsubscribeCampaignRepository;
        private final UnsubscribeAttemptRepository unsubscribeAttemptRepository;
        private final TriageGmailWriter triageGmailWriter;
        private final JdbcTemplate jdbcTemplate;
        private final Clock clock;

        public CampaignUndoService(... constructor inject all) { ... }

        @Transactional
        public CampaignUndoResult undo(UUID tenantId, UUID jobId) {
            // 1. Lookup campaign via job_id + tenant_id
            UUID campaignId = jdbcTemplate.queryForObject(
                "SELECT id FROM unsubscribe_campaign WHERE job_id = ? AND tenant_id = ?", UUID.class, jobId, tenantId);
            if (campaignId == null) throw new CampaignNotFoundException(jobId);

            UnsubscribeCampaignEntity campaign = unsubscribeCampaignRepository.findByIdAndTenantId(campaignId, tenantId)
                .orElseThrow(() -> new CampaignNotFoundException(campaignId));

            // 2. Window check (UNS-07b — 30d)
            if (campaign.getAppliedAt() == null) throw new IllegalStateException("Campaign not yet applied");
            Instant undoableUntil = UnsubscribeCampaignPolicy.undoableUntil(campaign.getAppliedAt());
            if (clock.instant().isAfter(undoableUntil)) {
                throw new UndoWindowExpiredException(campaignId, campaign.getAppliedAt(), UnsubscribeCampaignPolicy.UNDO_WINDOW);
            }
            if (campaign.getRevertedAt() != null) throw new IllegalStateException("Already reverted");

            // 3. List all archived gmail_message_id from triage_audit (recorded by UnsubscribeCampaignHandler Wave 4a)
            List<ArchivedRecord> records = jdbcTemplate.query(
                "SELECT gmail_message_id, ua.id AS attempt_id FROM triage_audit ta " +
                "JOIN unsubscribe_attempt ua ON ua.sender_email = ta.sender_email " +
                "WHERE ua.campaign_id = ? AND ta.action = 'ARCHIVE' AND ta.source = 'CLEANUP_CAMPAIGN' AND ta.reverted_at IS NULL",
                (rs, n) -> new ArchivedRecord(rs.getString("gmail_message_id"), (UUID) rs.getObject("attempt_id")),
                campaignId);

            // 4. For each archived message: TriageGmailWriter.restoreToInbox + removeLabel("Zero Mail/Unsubscribed")
            int restoredCount = 0;
            String labelId = triageGmailWriter.lookupLabelId(tenantId, UnsubscribeCampaignPolicy.UNSUBSCRIBED_LABEL_NAME);
            for (ArchivedRecord record : records) {
                triageGmailWriter.restoreToInbox(tenantId, record.gmailMessageId());
                triageGmailWriter.removeLabel(tenantId, record.gmailMessageId(), labelId);
                restoredCount++;
            }

            // 5. Mark revertedAt on campaign + audit rows
            Instant now = clock.instant();
            campaign.markReverted(now);
            unsubscribeCampaignRepository.save(campaign);
            jdbcTemplate.update("UPDATE triage_audit SET reverted_at = ? WHERE source = 'CLEANUP_CAMPAIGN' AND sender_email IN (SELECT sender_email FROM unsubscribe_attempt WHERE campaign_id = ?)", Timestamp.from(now), campaignId);

            log.info("event=cleanup_campaign_undone tenantId={} campaignId={} jobId={} restoredCount={}", tenantId, campaignId, jobId, restoredCount);
            return new CampaignUndoResult(campaignId, restoredCount);
        }

        public record CampaignUndoResult(UUID campaignId, int restoredCount) {}
        private record ArchivedRecord(String gmailMessageId, UUID attemptId) {}
    }
    ```
