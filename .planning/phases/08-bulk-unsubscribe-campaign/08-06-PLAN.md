    public class UnsubscribeDomainThrottle {
        private static final Logger log = LoggerFactory.getLogger(UnsubscribeDomainThrottle.class);
        private static final String KEY_60S = "throttle:unsubscribe:domain:%s:%s:60s";
        private static final String KEY_1H = "throttle:unsubscribe:domain:%s:%s:1h";

        private final StringRedisTemplate stringRedisTemplate;
        private final Clock clock;

        public UnsubscribeDomainThrottle(StringRedisTemplate stringRedisTemplate, Clock clock) {
            this.stringRedisTemplate = stringRedisTemplate;
            this.clock = clock;
        }

        public boolean acquire(UUID tenantId, String senderDomain) {
            String key60s = KEY_60S.formatted(tenantId, senderDomain);
            String key1h = KEY_1H.formatted(tenantId, senderDomain);

            Long count60s = stringRedisTemplate.opsForValue().increment(key60s);
            if (count60s != null && count60s == 1L) {
                stringRedisTemplate.expire(key60s, UnsubscribeCampaignPolicy.THROTTLE_PER_DOMAIN_WINDOW_60S);
            }
            Long count1h = stringRedisTemplate.opsForValue().increment(key1h);
            if (count1h != null && count1h == 1L) {
                stringRedisTemplate.expire(key1h, UnsubscribeCampaignPolicy.THROTTLE_PER_DOMAIN_WINDOW_1H);
            }

            if (count60s != null && count60s > UnsubscribeCampaignPolicy.THROTTLE_MAX_PER_DOMAIN_60S) {
                log.info("event=cleanup_throttle_deferred tenantId={} senderDomain={} window=60s count={}", tenantId, senderDomain, count60s);
                rollback(key60s, key1h);
                return false;
            }
            if (count1h != null && count1h > UnsubscribeCampaignPolicy.THROTTLE_MAX_PER_DOMAIN_1H) {
                log.info("event=cleanup_throttle_deferred tenantId={} senderDomain={} window=1h count={}", tenantId, senderDomain, count1h);
                rollback(key60s, key1h);
                return false;
            }
            return true;
        }

        private void rollback(String key60s, String key1h) {
            stringRedisTemplate.opsForValue().decrement(key60s);
            stringRedisTemplate.opsForValue().decrement(key1h);
        }
    }

       @Component
       public class ProcessingJobPurgeBatch {
           static final String LOCK_NAME = "processingJobPurge";
           private static final Logger log = LoggerFactory.getLogger(ProcessingJobPurgeBatch.class);
           private static final Duration PROCESSING_JOB_RETENTION = Duration.ofDays(90);
           private static final int BATCH_LIMIT = 1_000;

           private final JdbcTemplate jdbcTemplate;
           private final Clock clock;

           public ProcessingJobPurgeBatch(JdbcTemplate jdbcTemplate, Clock clock) {
               this.jdbcTemplate = jdbcTemplate;
               this.clock = clock;
           }

           @Scheduled(cron = "0 0 3 * * *", zone = "UTC")
           @SchedulerLock(name = LOCK_NAME, lockAtLeastFor = "PT1M", lockAtMostFor = "PT30M")
           public void scheduledPurge() {
               LockAssert.assertLocked();
               purge();
           }

           public int purge() {
               Instant cutoff = clock.instant().minus(PROCESSING_JOB_RETENTION);
               int totalDeleted = 0;
               int selectedCount;
               do {
                   PurgeResult result = purgeExpiredOnce(cutoff);
                   selectedCount = result.selectedCount();
                   totalDeleted += result.deletedCount();
               } while (selectedCount == BATCH_LIMIT);
               log.info("event=processing_job_purged totalDeleted={}", totalDeleted);
               return totalDeleted;
           }

           @Transactional
           public PurgeResult purgeExpiredOnce(Instant cutoff) {
               // CTE delete pattern mirror TriageAuditPurgeBatch.purgeExpiredOnce
               return jdbcTemplate.queryForObject(
                   "WITH expired_jobs AS (" +
                   "  SELECT id FROM processing_job " +
                   "  WHERE finished_at < ? AND status IN ('COMPLETED','FAILED') " +
                   "  ORDER BY finished_at ASC, id ASC " +
                   "  LIMIT ? FOR UPDATE SKIP LOCKED" +
                   "), deleted_jobs AS (" +
                   "  DELETE FROM processing_job WHERE id IN (SELECT id FROM expired_jobs) RETURNING id" +
                   ") SELECT " +
                   "  (SELECT COUNT(*) FROM expired_jobs) AS selected_count, " +
                   "  (SELECT COUNT(*) FROM deleted_jobs) AS deleted_count",
                   (rs, rowNum) -> new PurgeResult(rs.getInt("selected_count"), rs.getInt("deleted_count")),
                   Timestamp.from(cutoff), BATCH_LIMIT
               );
           }

           public record PurgeResult(int selectedCount, int deletedCount) {}
       }

       ```
       @Component
       public class ProcessingJobWorker {
           private static final Logger log = LoggerFactory.getLogger(ProcessingJobWorker.class);
           private static final Duration POLL_INTERVAL = Duration.ofSeconds(2);

           private final JdbcTemplate jdbcTemplate;
           private final TransactionTemplate transactionTemplate;
           private final UnsubscribeCampaignHandler unsubscribeCampaignHandler;
           private final Clock clock;
           private volatile boolean shouldRun = true;
           private Thread pollLoopThread;

           public ProcessingJobWorker(JdbcTemplate jdbcTemplate,
                                      PlatformTransactionManager transactionManager,
                                      UnsubscribeCampaignHandler unsubscribeCampaignHandler,
                                      Clock clock) {
               this.jdbcTemplate = jdbcTemplate;
               this.transactionTemplate = new TransactionTemplate(transactionManager);
               this.unsubscribeCampaignHandler = unsubscribeCampaignHandler;
               this.clock = clock;
           }

           @PostConstruct
           void startPolling() {
               this.pollLoopThread = Thread.ofVirtual().name("processing-job-worker").start(this::pollLoop);
           }

           @PreDestroy
           void stopPolling() throws InterruptedException {
               shouldRun = false;
               if (pollLoopThread != null) pollLoopThread.join(Duration.ofSeconds(5).toMillis());
           }

           private void pollLoop() {
               while (shouldRun) {
                   try {
                       Optional<ClaimedJob> claimed = transactionTemplate.execute(status -> claimNextJob());
                       if (claimed.isEmpty()) {
                           Thread.sleep(POLL_INTERVAL);
                           continue;
                       }
                       dispatchJob(claimed.get());
                   } catch (InterruptedException interrupted) {
                       Thread.currentThread().interrupt();
                       return;
                   } catch (RuntimeException unexpected) {
                       log.error("event=processing_job_worker_error", unexpected);
                       try { Thread.sleep(POLL_INTERVAL); } catch (InterruptedException ignored) { return; }
                   }
               }
           }

           private Optional<ClaimedJob> claimNextJob() {
               // SELECT ... FOR UPDATE SKIP LOCKED + UPDATE to RUNNING + heartbeat_at
               return jdbcTemplate.query(
                   "SELECT id, tenant_id, job_type, payload FROM processing_job " +
                   "WHERE status = 'QUEUED' AND next_run_at <= NOW() " +
                   "ORDER BY created_at FOR UPDATE SKIP LOCKED LIMIT 1",
                   resultSet -> {
                       if (!resultSet.next()) return Optional.<ClaimedJob>empty();
                       UUID id = (UUID) resultSet.getObject("id");
                       UUID tenantId = (UUID) resultSet.getObject("tenant_id");
                       String jobType = resultSet.getString("job_type");
                       String payload = resultSet.getString("payload");
                       jdbcTemplate.update(
                           "UPDATE processing_job SET status='RUNNING', started_at=NOW(), heartbeat_at=NOW(), updated_at=NOW() WHERE id = ?",
                           id);
                       log.info("event=processing_job_picked tenantId={} jobId={} jobType={}", tenantId, id, jobType);
                       return Optional.of(new ClaimedJob(id, tenantId, jobType, payload));
                   }
               );
           }

           private void dispatchJob(ClaimedJob claimed) {
               try {
                   ScopedValue.where(TenantContext.TENANT, claimed.tenantId().toString())
                              .run(() -> {
                                   switch (claimed.jobType()) {
                                       case "UNSUBSCRIBE_CAMPAIGN" -> unsubscribeCampaignHandler.handle(claimed.id(), claimed.tenantId(), claimed.payload());
                                       default -> throw new IllegalStateException("Unknown job_type: " + claimed.jobType());
                                   }
                              });
                   markCompleted(claimed.id());
               } catch (RuntimeException handlerFailure) {
                   log.error("event=processing_job_handler_failed jobId={}", claimed.id(), handlerFailure);
                   markFailed(claimed.id(), handlerFailure.getClass().getSimpleName());
               }
           }

           private void markCompleted(UUID jobId) {
               jdbcTemplate.update("UPDATE processing_job SET status='COMPLETED', finished_at=NOW(), updated_at=NOW() WHERE id = ?", jobId);
           }

           private void markFailed(UUID jobId, String failureReason) {
               jdbcTemplate.update("UPDATE processing_job SET status='FAILED', finished_at=NOW(), failure_reason=?, updated_at=NOW() WHERE id = ?", failureReason, jobId);
           }

           private record ClaimedJob(UUID id, UUID tenantId, String jobType, String payload) {}
       }
       ```

       Use `ScopedValue` binding to `TenantContext.TENANT` (Phase 1 — existing scoped value infra). Handler reads `TenantContext.currentTenantUuid()` inside dispatched logic.

           private final UnsubscribeAttemptRepository unsubscribeAttemptRepository;
           private final UnsubscribeHttpClient unsubscribeHttpClient;
           private final UnsubscribeMailtoSender unsubscribeMailtoSender;
           private final UnsubscribeDomainThrottle unsubscribeDomainThrottle;
           private final TriageGmailWriter triageGmailWriter;
           private final TriageAuditWriter triageAuditWriter;
           private final JdbcTemplate jdbcTemplate;
           private final ObjectMapper objectMapper;
           private final Clock clock;

           public void handle(UUID jobId, UUID tenantId, String payloadJson) {
               UnsubscribeCampaignPayload payload = parsePayload(payloadJson);  // {"campaignId": uuid, "schemaVersion": 1}
               UnsubscribeCampaignEntity campaign = unsubscribeCampaignRepository.findByIdAndTenantId(payload.campaignId(), tenantId)
                   .orElseThrow(() -> new IllegalStateException("Campaign not found: " + payload.campaignId()));
               campaign.markRunning() ... // set status RUNNING in campaign aggregate

               List<UnsubscribeAttemptEntity> attempts = unsubscribeAttemptRepository.findByCampaignIdOrderBySenderEmailAsc(campaign.getId());
               for (UnsubscribeAttemptEntity attempt : attempts) {
                   if (attempt.getState() != UnsubscribeAttemptState.PENDING) continue;  // skip already-processed (retry path)
                   executeSingleAttempt(jobId, tenantId, attempt);
                   updateHeartbeat(jobId);
               }
               campaign.markCompleted(clock.instant());
               unsubscribeCampaignRepository.save(campaign);
               log.info("event=cleanup_campaign_completed tenantId={} campaignId={} jobId={}", tenantId, campaign.getId(), jobId);
           }

           private void executeSingleAttempt(UUID jobId, UUID tenantId, UnsubscribeAttemptEntity attempt) {
               attempt.markRunning(clock.instant());
               unsubscribeAttemptRepository.save(attempt);

               // Throttle check — if denied, reschedule and skip
               if (!unsubscribeDomainThrottle.acquire(tenantId, attempt.getSenderDomain())) {
                   // Reschedule entire job 60s later — keeps attempt as PENDING (revert)
                   attempt.resetToPending();  // helper method on entity
                   unsubscribeAttemptRepository.save(attempt);
                   jdbcTemplate.update("UPDATE processing_job SET status='QUEUED', next_run_at = NOW() + INTERVAL '60 seconds', updated_at=NOW() WHERE id = ?", jobId);
                   throw new ThrottleDeferredException(attempt.getSenderDomain());  // caught by worker → no completion mark
               }

               // 1. Unsubscribe step
               UnsubscribeResult result = switch (attempt.getUnsubscribeMethod()) {
                   case ONE_CLICK -> {
                       URI url = lookupPersistedHttpsUrl(tenantId, attempt.getSenderEmail());
                       yield unsubscribeHttpClient.postOneClick(tenantId, url);
                   }
                   case MAILTO -> {
                       String rawMailto = lookupPersistedMailtoValue(tenantId, attempt.getSenderEmail());
                       String parsedRecipient = UnsubscribeMailtoUriParser.parse(rawMailto).recipient();
                       yield unsubscribeMailtoSender.sendUnsubscribeMailto(tenantId, rawMailto, parsedRecipient);
                   }
                   case NONE -> UnsubscribeResult.failed("NO_HEADER");  // shouldn't happen (filtered at preview)
               };

               // 2. Per-sender atomic gate
               if (result instanceof UnsubscribeResult.Failed failed) {
                   attempt.markFailed(failed.failureReason(), clock.instant());
                   unsubscribeAttemptRepository.save(attempt);
                   log.info("event=cleanup_campaign_step_failed tenantId={} senderDomain={} failureReason={}", tenantId, attempt.getSenderDomain(), failed.failureReason());
                   return;
               }

               // 3. Apply label + archive history (only if unsubscribe OK)
               int archivedCount = applyLabelAndArchiveHistory(tenantId, attempt.getSenderEmail());
               attempt.markOk(archivedCount, clock.instant());
               unsubscribeAttemptRepository.save(attempt);
               log.info("event=cleanup_campaign_step_ok tenantId={} senderDomain={} archivedCount={}", tenantId, attempt.getSenderDomain(), archivedCount);
           }

           private URI lookupPersistedHttpsUrl(UUID tenantId, String senderEmail) {
               // SELECT distinct list_unsubscribe_url FROM mail_message_observed WHERE tenant_id=? AND sender_email=? AND list_unsubscribe_url IS NOT NULL LIMIT 1
               // D-11 provenance check: only URL persisted in mail_message_observed accepted
               String url = jdbcTemplate.queryForObject(
                   "SELECT list_unsubscribe_url FROM mail_message_observed WHERE tenant_id=? AND sender_email=? AND list_unsubscribe_url IS NOT NULL LIMIT 1",
                   String.class, tenantId, senderEmail);
               return URI.create(url);
           }

           private String lookupPersistedMailtoValue(UUID tenantId, String senderEmail) {
               return jdbcTemplate.queryForObject(
                   "SELECT list_unsubscribe_mailto FROM mail_message_observed WHERE tenant_id=? AND sender_email=? AND list_unsubscribe_mailto IS NOT NULL LIMIT 1",
                   String.class, tenantId, senderEmail);
           }

           private int applyLabelAndArchiveHistory(UUID tenantId, String senderEmail) {
               // 1. Resolve or create label "Zero Mail/Unsubscribed"
               String labelId = triageGmailWriter.ensureLabelExists(tenantId, UnsubscribeCampaignPolicy.UNSUBSCRIBED_LABEL_NAME);
               // 2. List history Gmail message IDs from this sender (use existing GmailHistoryListReadService or direct Gmail API)
               List<String> messageIds = listHistoryMessageIds(tenantId, senderEmail);  // helper — delegate to existing
               int count = 0;
               for (String gmailMessageId : messageIds) {
                   triageGmailWriter.applyLabel(tenantId, gmailMessageId, labelId);
                   triageGmailWriter.archiveSkipInbox(tenantId, gmailMessageId);
                   triageAuditWriter.recordCleanupArchive(tenantId, gmailMessageId, ...);  // new method or reuse
                   count++;
               }
               return count;
           }

           private void updateHeartbeat(UUID jobId) {
               jdbcTemplate.update("UPDATE processing_job SET heartbeat_at = NOW(), updated_at = NOW() WHERE id = ?", jobId);
           }

           private UnsubscribeCampaignPayload parsePayload(String json) {
               try { return objectMapper.readValue(json, UnsubscribeCampaignPayload.class); }
               catch (IOException ioFailure) { throw new IllegalStateException("Malformed campaign payload", ioFailure); }
           }

           private record UnsubscribeCampaignPayload(UUID campaignId, int schemaVersion) {}  // D-19

           private static class ThrottleDeferredException extends RuntimeException {
               ThrottleDeferredException(String domain) { super("Throttle deferred: " + domain); }
           }
       }
