    public class UnsubscribeMailtoSender {

        public UnsubscribeResult sendUnsubscribeMailto(UUID tenantId, String rawMailtoValue, String persistedRecipientFromHeader) {
            try {
                UnsubscribeMailtoUriParser.ParsedMailto parsed = UnsubscribeMailtoUriParser.parse(rawMailtoValue);
                // D-06: validate recipient match persisted
                if (!parsed.recipient().equalsIgnoreCase(persistedRecipientFromHeader)) {
                    log.warn("event=cleanup_unsubscribe_mailto_recipient_mismatch tenantId={} senderDomain={}", tenantId, extractDomainFromEmail(parsed.recipient()));
                    return UnsubscribeResult.failed("MAILTO_RECIPIENT_MISMATCH");
                }
                MimeMessage mimeMessage = buildMimeMessage(tenantId, parsed);
                Message gmailMessage = encodeToGmailMessage(mimeMessage);
                Message sentMessage = gmailApiClientFactory.buildClientForTenant(tenantId)
                        .users().messages().send(USER_ID, gmailMessage).execute();
                log.info("event=cleanup_unsubscribe_mailto_sent tenantId={} senderDomain={}", tenantId, extractDomainFromEmail(parsed.recipient()));
                return UnsubscribeResult.ok(sentMessage.getId());
            } catch (IllegalArgumentException invalidUri) {
                log.warn("event=cleanup_unsubscribe_mailto_invalid_uri tenantId={} failureReason=MAILTO_INVALID_URI", tenantId);
                return UnsubscribeResult.failed("MAILTO_INVALID_URI");
            } catch (GoogleJsonResponseException googleFailure) {
                log.warn("event=cleanup_unsubscribe_mailto_gmail_failed tenantId={} statusCode={}", tenantId, googleFailure.getStatusCode());
                return UnsubscribeResult.failed("GMAIL_HTTP_" + googleFailure.getStatusCode());
            } catch (IOException networkFailure) {
                log.warn("event=cleanup_unsubscribe_mailto_network_failed tenantId={}", tenantId);
                return UnsubscribeResult.failed("NETWORK_ERROR");
            } catch (MessagingException mimeFailure) {
                log.warn("event=cleanup_unsubscribe_mailto_mime_failed tenantId={}", tenantId);
                return UnsubscribeResult.failed("MIME_BUILD_ERROR");
            }
        }
