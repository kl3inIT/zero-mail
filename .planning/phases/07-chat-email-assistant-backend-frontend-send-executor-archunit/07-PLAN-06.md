| Boundary | Description |

    ```ts
    import { useChat as useVercelChat } from '@ai-sdk/react';
    import { DefaultChatTransport } from 'ai';
    import { xsrfHeader } from '@/lib/http/xsrf';   // existing helper from rules-api / triage-api

    export function useChat({ chatId, initialMessages }: { chatId: string; initialMessages: UIMessage[] }) {
        return useVercelChat({
            id: chatId,
            initialMessages,
            experimental_throttle: 100,
            transport: new DefaultChatTransport({
                api: '/api/chat',
                credentials: 'include',
                // MEDIUM-5: CSRF header parity. Other typed API calls (features/rules/api/rules-api.ts,
                // features/triage/api/triage-api.ts) already send xsrfHeader(); without parity here,
                // POST /api/chat could 403 against the existing CSRF filter OR silently weaken posture.
                headers: () => xsrfHeader(),
            }),
            // DO NOT call reconnectToStream — vercel/ai#14027 (locked OFF in SPEC)
        });
    ```
