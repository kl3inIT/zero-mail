      @Component
      public class PairingCodeService {
          private static final int PAYLOAD_SIZE = 28; // 16 + 8 + 4
          private static final int SIG_TRUNCATED_SIZE = 16;
          private static final long TTL_SECONDS = 600;

          private final SecretKeySpec messagingLinkSecretKeySpec;
          private final SecureRandom secureRandom = new SecureRandom();
          private final TelegramProperties telegramProperties;

          public PairingCodeService(TelegramProperties telegramProperties) {
              this.telegramProperties = telegramProperties;
              byte[] secretBytes = Base64.getDecoder().decode(telegramProperties.messagingLinkSecret());
              if (secretBytes.length < 32) {
                  throw new IllegalStateException("messaging-link-secret must be 256-bit base64");
              }
              this.messagingLinkSecretKeySpec = new SecretKeySpec(secretBytes, "HmacSHA256");
          }

          public MintedPairingCode mint(UUID tenantId) {
              byte[] payload = ByteBuffer.allocate(PAYLOAD_SIZE)
                  .putLong(tenantId.getMostSignificantBits())
                  .putLong(tenantId.getLeastSignificantBits())
                  .putLong(secureRandom.nextLong())
                  .putInt((int)(Instant.now().getEpochSecond()))
                  .array();
              try {
                  Mac mac = Mac.getInstance("HmacSHA256");
                  mac.init(messagingLinkSecretKeySpec);
                  byte[] sigFull = mac.doFinal(payload);
                  byte[] sig = Arrays.copyOf(sigFull, SIG_TRUNCATED_SIZE);
                  Base64.Encoder urlEncoder = Base64.getUrlEncoder().withoutPadding();
                  String code = urlEncoder.encodeToString(payload) + "." + urlEncoder.encodeToString(sig);
                  Instant expiresAt = Instant.now().plusSeconds(TTL_SECONDS);
                  String deeplink = "https://t.me/" + telegramProperties.botUsername() + "?start=" + code;
                  return new MintedPairingCode(code, deeplink, expiresAt);
              } catch (NoSuchAlgorithmException | InvalidKeyException cryptoFailure) {
                  throw new IllegalStateException("HMAC-SHA256 unavailable", cryptoFailure);
              }
          }

          public ConsumedCode verify(String code) {
              String[] parts = code.split("\\.");
              if (parts.length != 2) throw new InvalidPairingCodeException();
              Base64.Decoder urlDecoder = Base64.getUrlDecoder();
              byte[] payload, signature;
              try {
                  payload = urlDecoder.decode(parts[0]);
                  signature = urlDecoder.decode(parts[1]);
              } catch (IllegalArgumentException decoderFailure) {
                  throw new InvalidPairingCodeException();
              }
              if (payload.length != PAYLOAD_SIZE || signature.length != SIG_TRUNCATED_SIZE) {
                  throw new InvalidPairingCodeException();
              }
              try {
                  Mac mac = Mac.getInstance("HmacSHA256");
                  mac.init(messagingLinkSecretKeySpec);
                  byte[] expectedSig = Arrays.copyOf(mac.doFinal(payload), SIG_TRUNCATED_SIZE);
                  if (!MessageDigest.isEqual(signature, expectedSig)) {
                      throw new InvalidPairingCodeException();
                  }
              } catch (NoSuchAlgorithmException | InvalidKeyException cryptoFailure) {
                  throw new IllegalStateException(cryptoFailure);
              }
              ByteBuffer buffer = ByteBuffer.wrap(payload);
              UUID tenantId = new UUID(buffer.getLong(), buffer.getLong());
              long nonce = buffer.getLong();
              int issuedAt = buffer.getInt();
              long now = Instant.now().getEpochSecond();
              if (now - issuedAt > TTL_SECONDS) throw new PairingCodeExpiredException();
              return new ConsumedCode(tenantId, nonce, Instant.ofEpochSecond(issuedAt));
          }

          public record MintedPairingCode(String code, String deeplink, Instant expiresAt) {}
          public record ConsumedCode(UUID tenantId, long nonce, Instant issuedAt) {}
      }
      ```

      @Component
      public class TelegramCommandRouter {
          private static final Pattern START_WITH_CODE = Pattern.compile("^/start\\s+(\\S+)$");
          private final PairingConsumeService pairingConsumeService;
          private final DisconnectTelegramService disconnectTelegramService;
          private final TelegramAccountRepository telegramAccountRepository;
          private final TelegramApiClient telegramApiClient;
          // ctor + log

          public boolean tryHandle(TelegramMessagePayload message) {
              String text = message.text();
              if (text == null || !text.startsWith("/")) return false;
              long chatId = message.chat().id();
              if (text.equals("/start") || text.startsWith("/start ")) {
                  handleStart(text, message);
                  return true;
              }
              if (text.equals("/help")) { sendHelp(chatId); return true; }
              if (text.equals("/disconnect")) { handleDisconnect(message); return true; }
              if (text.startsWith("/")) {
                  telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                      "Lệnh không hợp lệ. Gõ /help để xem các lệnh khả dụng.", "HTML", null));
                  return true;
              }
              return false;
          }

          private void handleStart(String text, TelegramMessagePayload message) {
              long chatId = message.chat().id();
              Matcher matcher = START_WITH_CODE.matcher(text);
              if (!matcher.matches()) {
                  telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                      "Chào! Hãy mở Zero Mail → Settings → Connected Apps để lấy mã kết nối.",
                      "HTML", null));
                  return;
              }
              String code = matcher.group(1);
              try {
                  pairingConsumeService.consume(code, chatId, message.from().id(),
                      message.from().username(), message.from().languageCode());
                  telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                      "✅ Đã kết nối thành công với Zero Mail!", "HTML", null));
              } catch (PairingCodeExpiredException expired) {
                  telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                      "Mã hết hạn. Hãy lấy mã mới trên Zero Mail web.", "HTML", null));
              } catch (InvalidPairingCodeException invalid) {
                  telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                      "Mã không hợp lệ.", "HTML", null));
              }
          }

          private void sendHelp(long chatId) {
              telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                  "Lệnh khả dụng:\n/start <code> — Kết nối\n/help — Trợ giúp\n/disconnect — Ngắt kết nối\n\n"
                      + "Privacy: Zero Mail không lưu nội dung email của bạn vào bot.", "HTML", null));
          }

          private void handleDisconnect(TelegramMessagePayload message) {
              long chatId = message.chat().id();
              telegramAccountRepository.findByTelegramChatId(chatId).ifPresentOrElse(
                  account -> {
                      disconnectTelegramService.disconnect(account.getTenantId());
                      telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                          "Đã ngắt kết nối. Mở Zero Mail web để kết nối lại bất cứ lúc nào.",
                          "HTML", null));
                  },
                  () -> telegramApiClient.sendMessage(new TelegramSendRequest(chatId,
                      "Tài khoản này chưa được liên kết.", "HTML", null)));
          }
      }
      ```

      public class TelegramBotInitializer implements ApplicationListener<ApplicationReadyEvent> {
          private final TelegramProperties telegramProperties;
          private final TelegramApiClient telegramApiClient;
          private final TelegramBotCommandsRegistry registry;
          private static final Logger logger = LoggerFactory.getLogger(TelegramBotInitializer.class);
