      public class TelegramWebhookIpAllowlistFilter extends OncePerRequestFilter {
          private static final Logger logger = LoggerFactory.getLogger(TelegramWebhookIpAllowlistFilter.class);
          private static final List<IpAddressMatcher> ALLOWED_TELEGRAM_CIDRS = List.of(
              new IpAddressMatcher("149.154.160.0/20"),
              new IpAddressMatcher("91.108.4.0/22"));

          @Override
          protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
                  throws ServletException, IOException {
              String remoteAddress = resolveSourceIp(request);
              boolean allowed = ALLOWED_TELEGRAM_CIDRS.stream().anyMatch(matcher -> matcher.matches(remoteAddress));
              if (!allowed) {
                  logger.warn("event=telegram.webhook.ip_rejected remoteIp_prefix={}", maskIp(remoteAddress));
                  response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
                  return;
              }
              chain.doFilter(request, response);
          }

          private static String resolveSourceIp(HttpServletRequest request) {
              // Honor X-Forwarded-For first IP if present (Pitfall 5 — VPS reverse proxy)
              String forwarded = request.getHeader("X-Forwarded-For");
              if (forwarded != null && !forwarded.isBlank()) {
                  return forwarded.split(",")[0].trim();
              }
              return request.getRemoteAddr();
          }

          private static String maskIp(String ip) {
              // Log only first 2 octets for IPv4 privacy
              String[] parts = ip.split("\\.");
              return parts.length == 4 ? parts[0] + "." + parts[1] + ".xx.xx" : "MASKED";
          }
      }
      ```

      public class TelegramWebhookSecretFilter extends OncePerRequestFilter {
          private static final Logger logger = LoggerFactory.getLogger(TelegramWebhookSecretFilter.class);
          private static final Pattern WEBHOOK_PATH_PATTERN = Pattern.compile("^/webhooks/telegram/([A-Za-z0-9_\\-]+)$");
          private final TelegramProperties telegramProperties;
          private final TelegramWebhookBadAttemptThrottle telegramWebhookBadAttemptThrottle;

          public TelegramWebhookSecretFilter(TelegramProperties telegramProperties,
                                             TelegramWebhookBadAttemptThrottle telegramWebhookBadAttemptThrottle) { ... }

          @Override
          protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
                  throws ServletException, IOException {
              Matcher pathMatcher = WEBHOOK_PATH_PATTERN.matcher(request.getRequestURI());
              if (!pathMatcher.matches()) {
                  response.setStatus(401);
                  return;
              }
              String urlSecretFromPath = pathMatcher.group(1);
              String headerSecret = request.getHeader("X-Telegram-Bot-Api-Secret-Token");
              boolean urlMatch = MessageDigest.isEqual(
                  urlSecretFromPath.getBytes(StandardCharsets.UTF_8),
                  telegramProperties.urlSecret().getBytes(StandardCharsets.UTF_8));
              boolean headerMatch = headerSecret != null && MessageDigest.isEqual(
                  headerSecret.getBytes(StandardCharsets.UTF_8),
                  telegramProperties.webhookSecret().getBytes(StandardCharsets.UTF_8));
              if (!urlMatch || !headerMatch) {
                  String sourceIp = resolveSourceIp(request);
                  boolean throttleAllowed = telegramWebhookBadAttemptThrottle.tryRecordBadAttempt(sourceIp);
                  response.setStatus(throttleAllowed ? 401 : 429);
                  logger.warn("event=telegram.webhook.unauthorized urlMatch={} headerMatch={} throttled={}",
                      urlMatch, headerMatch, !throttleAllowed);
                  return;
              }
              chain.doFilter(request, response);
          }

          private static String resolveSourceIp(HttpServletRequest request) { /* mirror IpAllowlistFilter */ }
      }
      ```
