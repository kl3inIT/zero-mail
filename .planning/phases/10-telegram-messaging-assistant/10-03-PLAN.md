      public class TelegramAccountUpsertRepository {

          public UUID consumeOrUpdateRelink(UUID tenantId, long telegramChatId, long telegramUserId,

              return jdbcTemplate.queryForObject(
                  """
                  INSERT INTO telegram_account
                    (id, tenant_id, telegram_chat_id, telegram_user_id, telegram_username,
                     language_code, status, notifications_enabled, notification_filter, linked_at)
                  VALUES (gen_random_uuid(), ?, ?, ?, ?, ?, 'CONNECTED', TRUE, '{}'::jsonb, NOW())
                  ON CONFLICT (tenant_id) DO UPDATE
                    SET status = 'CONNECTED',
                        telegram_chat_id = EXCLUDED.telegram_chat_id,
                        telegram_user_id = EXCLUDED.telegram_user_id,
                        telegram_username = EXCLUDED.telegram_username,
                        language_code = EXCLUDED.language_code,
                        linked_at = NOW(),
                        disconnected_at = NULL,
                        updated_at = NOW()
                  RETURNING id;
                  """,
