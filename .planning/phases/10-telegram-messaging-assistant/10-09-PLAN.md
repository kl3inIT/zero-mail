      ```java
      @Component
      @Scope(ConfigurableBeanFactory.SCOPE_PROTOTYPE)
      public class TelegramChatStreamSink implements ChatStreamSink {
          private final long telegramChatId;
          private final String languageCode;
          private final UUID tenantId;
          private final TelegramApiClient telegramApiClient;
          private final TelegramSendRateLimiter telegramSendRateLimiter;
          private final TelegramChatPreviewRenderer telegramChatPreviewRenderer;
          private final TelegramButtonLabels telegramButtonLabels;

          // Per-stream state
          private final AtomicReference<String> accumulatedText = new AtomicReference<>("");
          private final AtomicReference<Long> placeholderMessageId = new AtomicReference<>();
          private final AtomicReference<Sinks.Many<String>> textSink = new AtomicReference<>();
          private final AtomicReference<Disposable> samplerSubscription = new AtomicReference<>();

          public TelegramChatStreamSink(/* constructor injects */) { ... }

          @Override
          public void emitTextStart(String partId) {
              // D-05: send placeholder BEFORE first LLM chunk
              TelegramSendResponse response = telegramApiClient.sendMessage(new TelegramSendRequest(
                  telegramChatId, "✍️ Đang viết...", "HTML", null));
              placeholderMessageId.set(response.result().messageId());
              // Setup Reactor sink with sample cadence
              Sinks.Many<String> sink = Sinks.many().multicast().onBackpressureBuffer();
              textSink.set(sink);
              Disposable subscription = sink.asFlux()
                  .sample(Duration.ofMillis(800))         // D-06 — sample, not bufferTimeout
                  .filter(text -> !text.isEmpty())
                  .concatMap(text -> Mono.fromRunnable(() -> editMessageWithRetry(text)).subscribeOn(Schedulers.boundedElastic()))
                  .subscribe();
              samplerSubscription.set(subscription);
          }

          @Override
          public void emitTextDelta(String partId, String tokenText) {
              String current = accumulatedText.updateAndGet(prev -> prev + tokenText);
              Sinks.Many<String> sink = textSink.get();
              if (sink != null) sink.tryEmitNext(current);
          }

          @Override
          public void emitToolInputStart(String toolCallId, String toolName) {
              // Pitfall 8: do NOT accumulate tool args into text
          }

          @Override
          public void emitToolInputAvailable(String toolCallId, String toolName, String inputJson) {
              // For send/reply/forward/saveDraft — render preview card
              if (Set.of("sendEmail", "replyEmail", "forwardEmail", "saveDraft").contains(toolName)) {
                  telegramChatPreviewRenderer.renderPendingAction(telegramChatId, tenantId, toolCallId, toolName, inputJson, languageCode);
              }
          }

          @Override
          public void emitToolOutputAvailable(String toolCallId, String outputJson) {
              // Output sanitized by ToolOutputSanitizer (Phase 7) before arriving here.
              // For Telegram, render short summary only (NEVER body). Could noop and rely on next emitTextDelta to summarize.
          }

          @Override
          public void emitFinish(String reason) {
              Sinks.Many<String> sink = textSink.get();
              if (sink != null) sink.tryEmitComplete();
              Disposable sub = samplerSubscription.get();
              if (sub != null) sub.dispose();
              // Final edit — clear typing indicator
              String finalText = accumulatedText.get();
              if (placeholderMessageId.get() != null && !finalText.isEmpty()) {
                  editMessageWithRetry(finalText);
              }
          }

          @Override
          public void emitError(String code, String userFacingMessage) {
              if (placeholderMessageId.get() != null) {
                  try {
                      telegramApiClient.editMessageText(new TelegramEditRequest(
                          telegramChatId, placeholderMessageId.get(),
                          "Lỗi kết nối tạm thời, hãy thử lại.", "HTML", null));
                  } catch (Exception ignored) {}
              }
          }

          @Override
          public void emitHeartbeat() {
              // no-op for Telegram surface
          }

          @Override
          public void emitTextEnd(String partId) {
              // no-op — emitFinish handles final flush
          }

          @Override
          public void emitDataPersistence(UUID chatMessageId, String state) {
              // could correlate with assistant_pending_action via TelegramChatPreviewRenderer
          }

          private void editMessageWithRetry(String text) {
              Long messageId = placeholderMessageId.get();
              if (messageId == null) return;
              try {
                  telegramApiClient.editMessageText(new TelegramEditRequest(
                      telegramChatId, messageId, text, "HTML", null));
              } catch (TelegramRateLimit429Exception rate429) {
                  // D-07: pause + accumulate; single final edit after pause
                  telegramSendRateLimiter.notify429(telegramChatId, rate429.retryAfterSeconds());
                  // Don't rethrow — chunks continue accumulating in `accumulatedText`; the next sample window will retry naturally
              } catch (TelegramApiException apiFailure) {
                  // D-08: single retry with 500ms backoff
                  try {
                      Thread.sleep(500);
                      telegramApiClient.editMessageText(new TelegramEditRequest(
                          telegramChatId, messageId, text, "HTML", null));
                  } catch (Exception secondFailure) {
                      // Terminal: post error message + log; LLM stream NOT aborted
                      logger.warn("event=telegram.stream.transport_error tenantId={} chatId={} kind={}",
                          tenantId, telegramChatId, apiFailure.kind());
                      try {
                          telegramApiClient.editMessageText(new TelegramEditRequest(
                              telegramChatId, messageId, "Lỗi kết nối tạm thời, hãy thử lại.", "HTML", null));
                      } catch (Exception terminalFailure) { /* swallow */ }
                  } catch (InterruptedException ie) {
                      Thread.currentThread().interrupt();
                  }
              }
          }
      }
      ```
      Convention 5 logs: NO `text` content; only event + tenantId + chatId + kind.
