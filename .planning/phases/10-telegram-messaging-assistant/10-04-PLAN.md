    ```java
    package com.zeromail.core.messaging.telegram.gateway;

    @Component
    public final class TelegramSendRateLimiter {

        private final Bucket globalBucket;
        private final ConcurrentMap<Long, Bucket> perChatBuckets = new ConcurrentHashMap<>();
        private final ConcurrentMap<Long, Instant> chatPausedUntil = new ConcurrentHashMap<>();

        public TelegramSendRateLimiter() {
            this.globalBucket = Bucket.builder()
                .addLimit(Bandwidth.builder().capacity(30).refillIntervally(30, Duration.ofSeconds(1)).build())
                .build();
        }

        public boolean tryAcquire(long telegramChatId) {
            Instant pausedUntil = chatPausedUntil.get(telegramChatId);
            if (pausedUntil != null && Instant.now().isBefore(pausedUntil)) {
                return false;
            }
            Bucket perChat = perChatBuckets.computeIfAbsent(telegramChatId, id ->
                Bucket.builder()
                    .addLimit(Bandwidth.builder().capacity(1).refillIntervally(1, Duration.ofSeconds(1)).build())
                    .build());
            return perChat.tryConsume(1) && globalBucket.tryConsume(1);
        }

        public void notify429(long telegramChatId, int retryAfterSeconds) {
            chatPausedUntil.put(telegramChatId, Instant.now().plusSeconds(retryAfterSeconds));
        }

        // Memory leak prevention — Plan 04 ships baseline; future plan may add @Scheduled eviction
        public void evictInactive(long telegramChatId) {
            perChatBuckets.remove(telegramChatId);
            chatPausedUntil.remove(telegramChatId);
        }
    }
    ```

      ```java
      @Component
      public class TelegramApiClient {
          private final RestClient restClient;
          private final TelegramProperties telegramProperties;
          private final TelegramSendRateLimiter telegramSendRateLimiter;
          private final ObjectMapper objectMapper;
          private static final Logger logger = LoggerFactory.getLogger(TelegramApiClient.class);

          public TelegramApiClient(RestClient.Builder restClientBuilder,
                                   TelegramProperties telegramProperties,
                                   TelegramSendRateLimiter telegramSendRateLimiter,
                                   ObjectMapper objectMapper) {
              this.restClient = restClientBuilder
                  .baseUrl(telegramProperties.apiBaseUrl().toString())
                  .build();
              this.telegramProperties = telegramProperties;
              this.telegramSendRateLimiter = telegramSendRateLimiter;
              this.objectMapper = objectMapper;
          }

          public TelegramSendResponse sendMessage(TelegramSendRequest request) {
              if (!telegramSendRateLimiter.tryAcquire(request.chatId())) {
                  throw new TelegramApiException(TelegramErrorKind.RATE_LIMIT_429,
                      new IllegalStateException("Local rate limit hit"));
              }
              return execute("sendMessage", request, TelegramSendResponse.class, request.chatId());
          }

          public TelegramSendResponse editMessageText(TelegramEditRequest request) {
              if (!telegramSendRateLimiter.tryAcquire(request.chatId())) {
                  throw new TelegramApiException(TelegramErrorKind.RATE_LIMIT_429,
                      new IllegalStateException("Local rate limit hit"));
              }
              return execute("editMessageText", request, TelegramSendResponse.class, request.chatId());
          }

          public void editMessageReplyMarkup(...) { ... }
          public void answerCallbackQuery(TelegramAnswerCallbackRequest request) { ... }
          public void setMyCommands(List<Map<String, String>> commands) { ... }
          public List<Map<String, String>> getMyCommands() { ... }
          public void setMyProfilePhoto(byte[] photoBytes) { ... }

          private <T> T execute(String method, Object body, Class<T> responseType, long chatId) {
              String botPath = "/bot" + telegramProperties.botToken() + "/" + method;
              try {
                  return restClient.post().uri(botPath)
                      .contentType(MediaType.APPLICATION_JSON)
                      .body(body)
                      .retrieve()
                      .body(responseType);
              } catch (RestClientResponseException restException) {
                  int status = restException.getStatusCode().value();
                  if (status == 429) {
                      int retryAfter = parseRetryAfterFromBody(restException);
                      telegramSendRateLimiter.notify429(chatId, retryAfter);
                      throw new TelegramRateLimit429Exception(retryAfter, restException);
                  }
                  throw new TelegramApiException(mapStatus(status, restException), restException);
              } catch (ResourceAccessException raException) {
                  throw new TelegramApiException(
                      isTimeout(raException) ? TelegramErrorKind.TIMEOUT : TelegramErrorKind.NETWORK_ERROR,
                      raException);
              }
          }

          private int parseRetryAfterFromBody(RestClientResponseException restException) {
              try {
                  JsonNode body = objectMapper.readTree(restException.getResponseBodyAsString());
                  JsonNode retryAfter = body.path("parameters").path("retry_after");
                  return retryAfter.isInt() ? retryAfter.asInt() : 30; // fallback 30s
              } catch (Exception parseFailure) {
                  return 30;
              }
          }

          private TelegramErrorKind mapStatus(int status, RestClientResponseException restException) {
              return switch (status) {
                  case 401 -> TelegramErrorKind.UNAUTHORIZED;
                  case 403 -> {
                      String body = restException.getResponseBodyAsString();
                      yield body.contains("bot was blocked") ? TelegramErrorKind.FORBIDDEN_USER_BLOCKED
                                                              : TelegramErrorKind.BAD_REQUEST;
                  }
                  case 400 -> TelegramErrorKind.BAD_REQUEST;
                  case 404 -> TelegramErrorKind.UNKNOWN_CHAT;
                  default -> status >= 500 ? TelegramErrorKind.UNKNOWN_5XX : TelegramErrorKind.BAD_REQUEST;
              };
          }
      }
      ```
