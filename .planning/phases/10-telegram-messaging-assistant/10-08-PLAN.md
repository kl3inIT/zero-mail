      @Component
      public class MessagingNotificationProcessor {
          private final TelegramApiClient telegramApiClient;
          private final TelegramAccountRepository telegramAccountRepository;
          private final TelegramNotificationLogRepository telegramNotificationLogRepository;
          private final TelegramButtonLabels telegramButtonLabels;
          private final TelegramCallbackTokenService telegramCallbackTokenService;
          private final ProcessingJobEnqueuer processingJobEnqueuer;
          private final ProcessingJobRepository processingJobRepository;
          private final ObjectMapper objectMapper;
          private static final Logger logger = LoggerFactory.getLogger(MessagingNotificationProcessor.class);

          @Transactional
          public void handle(UUID jobId, UUID tenantId, String payloadJson) {
              JsonNode root = parsePayload(payloadJson);
              long telegramChatId = root.path("telegramChatId").asLong();
              String notificationKind = root.path("notificationKind").asText();
              JsonNode payload = root.path("payload");
              String gmailMessageId = payload.path("gmailMessageId").asText();

              TelegramAccountEntity account = telegramAccountRepository.findByTelegramChatId(telegramChatId)
                  .orElseThrow(() -> new IllegalStateException("No account for chatId"));

              String text = renderNotificationText(payload, account.getLanguageCode());
              long epochMinute = Instant.now().getEpochSecond() / 60;
              // Compute action tokens for inline_keyboard
              String token16 = telegramCallbackTokenService.computeInlineToken(
                  "archive", tenantId, gmailMessageId, epochMinute);
              Map<String,Object> replyMarkup = telegramButtonLabels.buildPrimaryKeyboard(
                  gmailMessageId, token16, account.getLanguageCode());

              try {
                  TelegramSendResponse response = telegramApiClient.sendMessage(
                      new TelegramSendRequest(telegramChatId, text, "HTML", replyMarkup));
                  // Insert log row (no body content)
                  TelegramNotificationLogEntity logRow = new TelegramNotificationLogEntity(
                      UUID.randomUUID(), tenantId, telegramChatId, gmailMessageId, notificationKind,
                      response.result() != null ? response.result().messageId() : null,
                      Instant.now(), null, null);
                  telegramNotificationLogRepository.save(logRow);
                  logger.info("event=telegram.notification.sent tenantId={} chatId={}", tenantId, telegramChatId);
              } catch (TelegramRateLimit429Exception rate429) {
                  processingJobEnqueuer.rescheduleFor429(jobId, rate429.retryAfterSeconds());
                  logger.warn("event=telegram.notification.rate_limited tenantId={} retryAfter={}",
                      tenantId, rate429.retryAfterSeconds());
                  // Don't bump admin_requeue_count per CONTEXT D-04
              } catch (TelegramApiException apiFailure) {
                  if (apiFailure.kind() == TelegramErrorKind.FORBIDDEN_USER_BLOCKED) {
                      account.setStatus(TelegramAccountStatus.BLOCKED);
                      account.setBlockedAt(Instant.now());
                      telegramAccountRepository.save(account);
                      logger.warn("event=telegram.notification.user_blocked tenantId={}", tenantId);
                      // Mark job as DEAD_LETTER — no retry
                      throw new MessagingDeliveryDeadException(JobFailureReason.TELEGRAM_FORBIDDEN_BLOCKED, apiFailure);
                  }
                  throw new MessagingDeliveryFailedException(mapToFailureReason(apiFailure.kind()), apiFailure);
              }
          }
