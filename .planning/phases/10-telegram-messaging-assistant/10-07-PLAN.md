      public class TelegramCallbackTokenService {

           */
          public String computeInlineToken(String actionId, UUID tenantId, String resourceId, long messageSentEpochMinute) {

          }

          /**
           * Chat assistant confirm token (D-02).
           * Format: sha256(actionType + ":" + chatMessageId + ":" + toolCallId).hex.substring(0, 16)
           */
          public String computeAssistantToken(String actionType, UUID chatMessageId, String toolCallId) {
              String input = actionType + ":" + chatMessageId + ":" + toolCallId;
              return sha256Hex(input).substring(0, 16);
          }

          public boolean verifyConstantTime(String provided, String expected) {
              if (provided == null || expected == null) return false;
              return MessageDigest.isEqual(provided.getBytes(StandardCharsets.UTF_8),
                                            expected.getBytes(StandardCharsets.UTF_8));
          }

          private static String sha256Hex(String input) {
              try {
                  MessageDigest md = MessageDigest.getInstance("SHA-256");
                  byte[] hash = md.digest(input.getBytes(StandardCharsets.UTF_8));
                  StringBuilder hex = new StringBuilder();
                  for (byte b : hash) hex.append(String.format("%02x", b));
                  return hex.toString();
              } catch (NoSuchAlgorithmException nsae) { throw new IllegalStateException(nsae); }
          }
      }

      @Component
      public class TelegramCallbackRouter {
          private final TelegramAccountRepository telegramAccountRepository;
          private final TelegramApiClient telegramApiClient;
          private final TelegramInlineButtonHandler telegramInlineButtonHandler;
          private final TelegramAssistantConfirmHandler telegramAssistantConfirmHandler;
          private static final Logger logger = LoggerFactory.getLogger(TelegramCallbackRouter.class);

          public void route(TelegramCallbackPayload callback) {
              if (!callback.message().chat().isPrivate()) {
                  rejectNonPrivate(callback);
                  return;
              }
              long chatId = callback.message().chat().id();
              Optional<TelegramAccountEntity> accountOpt = telegramAccountRepository.findByTelegramChatId(chatId);
              if (accountOpt.isEmpty()) {
                  telegramApiClient.answerCallbackQuery(new TelegramAnswerCallbackRequest(callback.id(),
                      "Tài khoản chưa được liên kết.", true));
                  return;
              }
              TelegramAccountEntity account = accountOpt.get();
              if (callback.from().id() != account.getTelegramUserId()) {
                  telegramApiClient.answerCallbackQuery(new TelegramAnswerCallbackRequest(callback.id(),
                      "Bạn không có quyền xác nhận draft này", true));
                  logger.warn("event=telegram.callback.unauthorized_actor tenantId={} chatId={} fromId={}",
                      account.getTenantId(), chatId, callback.from().id());
                  return;
              }
              String data = callback.data();
              if (data == null) { acknowledgeEmpty(callback); return; }
              String[] parts = data.split(":");
              if (parts.length < 3) { acknowledgeEmpty(callback); return; }
              String actionId = parts[0];
              // Dispatch: assistant chat confirm vs inline-button triage notification
              if (Set.of("send", "reply", "forward", "save_draft").contains(actionId)) {
                  telegramAssistantConfirmHandler.handle(callback, account, actionId, parts);
              } else {
                  telegramInlineButtonHandler.handle(callback, account, actionId, parts);
              }
          }

          private void rejectNonPrivate(TelegramCallbackPayload callback) {
              telegramApiClient.answerCallbackQuery(new TelegramAnswerCallbackRequest(callback.id(),
                  "Vui lòng dùng bot trong DM.", true));
          }

          private void acknowledgeEmpty(TelegramCallbackPayload callback) {
              telegramApiClient.answerCallbackQuery(new TelegramAnswerCallbackRequest(callback.id(), "", false));
          }
      }
