      package com.zeromail.core.chat.domain;

      import com.zeromail.core.shared.lang.IdentifiedEnum;
      import java.util.NoSuchElementException;
      import java.util.stream.Stream;

      /**
       * Surface that the chat assistant streams its response into.
       * <ul>
       *   <li>WEB_SSE — existing Server-Sent Events to apps/web (Phase 7)</li>
       *   <li>TELEGRAM — Telegram editMessageText cadence via TelegramChatStreamSink (Phase 10 TG-16)</li>
       * </ul>
       */
      public enum ResponseSurface implements IdentifiedEnum {
          WEB_SSE,
          TELEGRAM;

          @Override
          public String id() {
              return name();
          }

          public static ResponseSurface fromId(String id) {
              return Stream.of(values())
                      .filter(value -> value.id().equals(id))
                      .findFirst()
                      .orElseThrow(() -> new NoSuchElementException("Unknown ResponseSurface id: " + id));
          }
      }
