     package com.zeromail.api;

     import ch.qos.logback.classic.Logger;
     import ch.qos.logback.classic.spi.ILoggingEvent;
     import ch.qos.logback.core.read.ListAppender;
     import com.zeromail.api.security.TestSessionSupport;
     import com.zeromail.core.persistence.*;
     import com.zeromail.core.support.PostgresContainerTest;
     import java.util.List;
     import java.util.UUID;
     import org.junit.jupiter.api.*;
     import org.slf4j.LoggerFactory;
     import org.springframework.beans.factory.annotation.Autowired;
     import org.springframework.boot.test.context.SpringBootTest;
     import org.springframework.boot.test.web.client.TestRestTemplate;
     import org.springframework.context.annotation.Import;
     import org.springframework.http.HttpEntity;
     import org.springframework.http.HttpHeaders;
     import org.springframework.http.HttpMethod;
     import org.springframework.http.MediaType;
     import org.springframework.test.context.ActiveProfiles;
     import static org.assertj.core.api.Assertions.*;

     @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
     @ActiveProfiles("test")
     @Import(TestSessionSupport.class)
     class LogScrubSyntheticTrafficTest extends PostgresContainerTest {
         // Sentinel values planted into seed data. If any of these ever appear in a log line,
         // the log-safety contract (plan 03) is violated. Chosen to be obviously non-PII and
         // easy to grep, per VALIDATION.md Manual-Only Verifications row for FND-03.
         private static final String LEAK_PROBE_SUBJECT = "leak-probe-12345";
         private static final String LEAK_PROBE_EMAIL = "leak-probe@example.test";
         private static final String LEAK_PROBE_REFRESH_TOKEN = "LEAK-REFRESH-TOKEN-ABC";

         @Autowired TestRestTemplate rest;
         @Autowired TenantRepository tenants;
         @Autowired UserRepository users;
         @Autowired GmailConnectionRepository conns;
         @Autowired TestSessionSupport.TestSessionMinter minter;

         Logger root;
         ListAppender<ILoggingEvent> appender;

         @BeforeEach void setUp() {
             root = (Logger) LoggerFactory.getLogger(Logger.ROOT_LOGGER_NAME);
             appender = new ListAppender<>();
             appender.start();
             root.addAppender(appender);
         }

         @AfterEach void tearDown() {
             appender.stop();
             root.detachAppender(appender);
         }

         @Test
         void real_request_traffic_never_leaks_sensitive_content() {
             // --- Seed tenant + user + gmail connection carrying the sentinel values ---
             UUID tenantId = UUID.randomUUID();
             tenants.save(new TenantEntity(tenantId, "leak-probe-tenant"));
             var user = ScopedValue.where(com.zeromail.core.tenant.TenantContext.TENANT, tenantId.toString())
                 .call(() -> users.save(new UserEntity(
                     UUID.randomUUID(), tenantId, LEAK_PROBE_SUBJECT, LEAK_PROBE_EMAIL)));
             ScopedValue.where(com.zeromail.core.tenant.TenantContext.TENANT, tenantId.toString())
                 .run(() -> {
                     var gc = new GmailConnectionEntity();
                     // Set required fields via setters (see plan 04): id, tenantId, googleEmail, status,
                     // and refresh_token_encrypted bytes. We store the sentinel plaintext directly into
                     // the byte[] here (test-only) so that any naive toString / debug log of the entity
                     // would surface it. In production, plan 06's cipher writes an AES-GCM envelope.
                     gc.setRefreshTokenEncrypted(LEAK_PROBE_REFRESH_TOKEN.getBytes(java.nio.charset.StandardCharsets.UTF_8));
                     conns.save(gc);
                 });

             // --- Mint a real session cookie for the seeded user ---
             String cookie = minter.mint(LEAK_PROBE_SUBJECT, LEAK_PROBE_EMAIL);
             HttpHeaders headers = new HttpHeaders();
             headers.add("Cookie", cookie);
             headers.setContentType(MediaType.APPLICATION_JSON);

             // --- Drive real authenticated requests through Phase 1 endpoints ---
             rest.exchange("/me", HttpMethod.GET, new HttpEntity<>(headers), String.class);
             rest.exchange("/tenant/status", HttpMethod.GET, new HttpEntity<>(headers), String.class);
             rest.exchange("/onboarding/select-template", HttpMethod.POST,
                 new HttpEntity<>("{\"templateKey\":\"archive-receipts\"}", headers), String.class);

             // --- Render the captured log stream and grep for sentinel values ---
             List<String> capturedLines = appender.list.stream()
                 .map(ev -> ev.getFormattedMessage() + " " + ev.getMDCPropertyMap())
                 .toList();
             String combined = String.join("\n", capturedLines);

             assertThat(combined)
                 .as("FND-03: no raw email body / prompt / completion / refresh token leaks")
                 .doesNotContain(LEAK_PROBE_SUBJECT)
                 .doesNotContain(LEAK_PROBE_REFRESH_TOKEN);

             // Additional well-known sensitive field names must not leak either.
             assertThat(combined)
                 .doesNotContain("refresh_token=")
                 .doesNotContain("\"body\":")
                 .doesNotContain("\"prompt\":")
                 .doesNotContain("\"completion\":");

             // Proves the plan 03 TurboFilter actually fired on a real request path (not just
             // in an in-process debug log line). At least one log event should carry the
             // `scrubbed=true` MDC marker emitted by SensitiveMarkerScrubFilter.
             assertThat(appender.list)
                 .as("at least one log event carries scrubbed=true MDC key")
                 .anySatisfy(ev -> assertThat(ev.getMDCPropertyMap()).containsEntry("scrubbed", "true"));
         }
     }
     ```

     # Zero Mail — Privacy Policy (Draft)

     ## What Zero Mail does with your Gmail data
     - We request the Gmail `modify` scope only to label, archive, or save drafts based on rules you configure. We never auto-send.
     - We do **not** store raw email bodies, LLM prompts, LLM completions, or embeddings long-term. Message content is held only in short-lived in-memory caches during triage and is never persisted.
     - Metadata we do store: sender, subject, thread id, triage action, rule id, timestamp. Enough to support audit + undo (Phase 4) and analytics (Phase 5) — not enough to reconstruct message content.

     ## Tokens and encryption
     - OAuth refresh tokens are encrypted at rest with AES-GCM-256; keys live in GCP Secret Manager and never touch the database.

     ## User control
     - Disconnect Gmail: removes the grant; no further API calls are made on your behalf.
     - Delete account: cascades to all rows in tenants, users, gmail_connections, and onboarding_selections.
     - Revoke at any time from your Google Account's Third-Party Apps page; if Google returns `invalid_grant` on the next refresh, Zero Mail flips your connection to DISCONNECTED and prompts you to reconnect.

     # Scopes Justification — https://www.googleapis.com/auth/gmail.modify

     ## Why this scope (minimum necessary)
     Zero Mail's core function is automated triage: labeling, archiving, and drafting replies based on user-authored rules. These actions require modify-scope; read-only would leave archive and label unavailable and force users into a supervisory role that defeats the product.

     ## What we do NOT do with this scope
     - No sending. Auto-send is prohibited at the gateway layer (TRG-03, Phase 4) and enforced by an ArchUnit ban on the `SEND` action.
     - No long-term storage of email bodies. Body content touches only short-lived in-memory caches during LLM calls (Phase 2C) and is discarded after each request.
     - No extraction of content beyond metadata needed for audit + analytics.

     ## Data handling posture
     - Refresh tokens encrypted with AES-GCM-256, key via GCP Secret Manager (Phase 1, plan 06).
     - Tenant isolation enforced by ScopedValue + Hibernate `@TenantId` + ArchUnit ban on ThreadLocal + concurrent leak test (Phase 1, plans 02, 05).
     - Log safety contract: `@Sensitive<T>` wrapper + Logback TurboFilter + ArchUnit rules preventing body/prompt/completion from reaching logs (Phase 1, plan 03).

     ## Retention / deletion / revocation (AUTH-03, AUTH-05)
     - Account-delete cascades to all tenant-owned tables (Phase 1, plan 07).
     - **AUTH-05** — Externally-revoked grants surface as DISCONNECTED on the next request: Spring Security's refresh-token client is wrapped by `DisconnectDetectingRefreshTokenClient`; when Google returns `invalid_grant`, an `OAuth2TokenRefreshFailed` event is published, `GmailAccessGuard` flips `gmail_connections.status` to DISCONNECTED, and the user sees a reconnect prompt in `/settings` (Phase 1, plans 05 and 08). This means a user revoking the grant in their Google Account dashboard does not require Zero Mail to poll — the state synchronizes lazily on the next outbound token refresh.
     ```
